	}

	desc = "The East African campaign has ended or hardened into a settlement. Parliament now judges India's sanctions, mediation or observer mission by the result on the ground, and the Indian Ocean staff incorporates the diplomatic and operational lessons."

	offset = 15
	deathdate = { day = 29 month = december year = 1964 }

		name = "Enter the African lessons in national policy"
