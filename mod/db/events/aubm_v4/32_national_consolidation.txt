# The unrestricted country catalogue remains available for players who want a
# sandbox campaign before the 1937 strategic phase, but it must be enabled
# deliberately.  The opt-in carries no reward and cannot bypass legal-target
# or commitment guards.
event = {
	id = 9281008
	random = no
	persistent = yes
	country = IND

	decision = {
		flag = ind_v43_war_cabinet
		flag = ind_v3_integrated
		flag = aubm_v4_union_register_opened
		atwar = no
		NOT = { year = 1937 }
		NOT = { flag = ind_aubm_unrestricted_sandbox }
	}
	decision_trigger = {
		flag = ind_v43_war_cabinet
		flag = ind_v3_integrated
		flag = aubm_v4_union_register_opened
		atwar = no
		NOT = { year = 1937 }
		NOT = { flag = ind_aubm_unrestricted_sandbox }
	}

	name = "Enable Unrestricted Early Campaigns"
	decision_desc = "Optional permanent sandbox opt-in. This exposes the War Cabinet and arbitrary declarations before 1937. It grants no reward, does not waive alliance or legal-target guards, and can close peacetime development chains by starting an early war."
	desc = "India's constitutional programme is designed to run at peace through 1936. Enabling the unrestricted catalogue early is for alternate starts and deliberate sandbox wars; it is not required for the normal campaign."
	style = 2
	picture = "aubm_v4_grand_strategy"

	date = { day = 0 month = january year = 1933 }
	offset = 5
	deathdate = { day = 29 month = december year = 1964 }

	action_a = {

	}
}

	country = IND

		}

		flag = ind_v43_war_cabinet

			year = 1937
